رہائش کیئر پیٹھ غور کرُن ہیکِہ سکون، غم، غیر یقینی یا واریاہ احساس أکِس وِز آنِتھ۔ یہِ فٲصلہ چھُ تِژ مدد لبنہ مُتعلق یۄسہٕ شخص سٕنٛزن ضرورتن تہٕ ترجیحن سٍتۍ میل کھایِہ۔ یہِ چھُنہ کانٛہہ امتحان زِ خاندان کٔژاہ چھُ خیال تھوان، تہٕ نہ چھُ یہِ کانٛہہ مقرر قدم یَتھ پیٹھ پرٛتھ کانٛہہ أکِس وقتس پیٹھ واتان چھُ۔

[Alzheimer's Association ہُند طویل مدتی دیکھ بھال رہنما](https://www.alz.org/help-support/caregiving/care-options/long-term-care)، تِہُند [رہائش کیئر چیک لسٹ](https://www.alz.org/getmedia/2f238875-3f5d-447e-adcc-dd9fc07f828b/alzheimers-dementia-choosing-residential-care-ts.pdf) تہٕ [Dementia Australia ہٕنز رہائش کیئر رہنمائی](https://www.dementia.org.au/living-dementia/care-options/residential-care) چھِ شخص سٕنٛزن ضرورتہٕ، عملہ، سرگرمیہٕ، ماحول تہٕ خرچ پیٹھ نظر تھاونٕچ صلاح دِوان۔ یہِ صفہٕ چھُنہ کُنہِ خاص گھرٕچ سفارش کران تہٕ نہ چھُ مقامی پیشہ ورانہ یا مالی مشورٕچ جای لینہ۔

## مےٚ کِتھ کٔنۍ گژِھ پتا زِ رہائش کیئر پیٹھ غور کرُن چھا مناسب؟

وُچھِو زِ وَنہ شخصس کٔم مدد چھِ ضرورت تہٕ کیا سۄ مدد گرِ منز ہیکِہ مِلِتھ۔ شخص سٕنز حفاظت، روزمرٕ دیکھ بھال ضرورتہٕ، سماجی زندگی تہٕ ترجیحہٕ پیٹھ کٔرِو غور۔ یہِ تہِ وُچھِو زِ کیا سنبھال کرنٕچ ذِمہ دٲری چھا شخص سٕنٛزِ یا تِہٕنز سنبھال کرن والین ہٕنٛزِ صحتس پیٹھ اثر تراوان۔

کانٛہہ اکھ مرحلہ یا فہرست چھَنہ یۄسہٕ یہِ طے کرِ زِ کٲنٛسہِ گژِھ کَمہِ وقتہٕ پکُن۔ اکھ شخص ہیکِہ کمیونٹی ماحول پسند کٔرِتھ، یا ہیکِہ اضافی مدد سان گرِ منز روزُن یژِھتھ۔ شخصس سٍتۍ کتھ کٔرِو، تہٕ اگر فائدہٕ مند آسِہ، تِہٕنٛدِس ڈاکٹرس، سوشل ورکر سٍتۍ یا کیئر کوآرڈینیٹر سٍتۍ۔ پکُنہ برونٛہہ پٕرژھِو زِ کٔم قسمٕچ مدد ہیکِہ موزوں ٲسِتھ۔

فٲصلن منز کٔرِو شخصس تیوٗت شٲمِل یِیوٗت سُہ چاہِ۔ پٕرژھِو زِ تِہِندِ خٲطرٕ کیا چھُ اہم: جایِہ، کھین، زبان، ثقافت، مذہبی عمل، مانوس سرگرمیہٕ یا خاندانس تہٕ دوستن ہٕنز ملاقات۔ تِہٕنز ترجیحہٕ ہیکَن فہرست چھوٹی کرنس منز مدد کٔرِتھ، اگرچہ کیہنہ دیکھ بھال ضرورتہٕ بدلِتھ تہِ آسَن۔

## گرن ہُند دورہ کرنہ برونٛہہ مےٚ کیا گژِھ معلوم کرُن؟

گۄڈٕ بنٲوِو اکھ چھوٹی فہرست تِمَن مددن ہٕنز یِم شخصس ضرورت چھِ۔ اتھ منز ہیکِہ روزمرٕ کامین منز مدد، نرسنگ، سماجی رابطہ، مانوس زبان یا نبر وقت گزارنٕچ جایِہ شٲمِل ٲسِتھ۔ اگر تُہؠ یقین آسِہ نہ، تِمہِ صحت یا سماجی دیکھ بھال پیشہ ورَس پٕرژھِو زِ کٔمہِ خدمتن ہٕنز ضرورت ہیکِہ پٮ۪ٹھ آسِتھ۔

دورہ کرنہ برونٛہہ کٔرِو گرن فون۔ پٕرژھِو زِ کیا تِم ہیکَن شخص سٕنٛزن ضرورتن ہٕنز مدد کٔرِتھ، کیا کانٛہہ کمرہ دستیاب ہیکِہ ٲسِتھ، برونٛہہ کُن قدم کیا چھِ تہٕ تِمن تُہندِ طرفہٕ کٔم معلومات چھِ ضرورت۔ خدمتن تہٕ فیسن مُتعلق تحریری معلومات منگِو۔ اہلیت، فنڈنگ تہٕ لائسنس قواعد چھِ جایِہ مطٲبق بدلِوان، لہٰذا عام مضمون پیٹھ بھروسہ کرنہ بجایِہ کٔرِو مقامی ذرائع چیک۔

پٕرژھِو زِ کیا عملس چھا ڈیمنشیا دیکھ بھالچ تربیت، دیکھ بھال منصوبہ کِتھ کٔنۍ چھِ بنان یِوان تہٕ خاندانہٕ کِتھ کٔنۍ ہیکَن نوِ معلومات شیئر کٔرِتھ۔ معلوم کٔرِو زِ روزمرٕ معمولات کِتھ کٔنۍ چھِ چلان، کٔمہِ سرگرمیہٕ چھِ پیش یِوان کرنہ تہٕ باشندن ہٕنز پسند چُننس منز کِتھ کٔنۍ چھِ مدد یِوان کرنہ۔ میموری کیئر یونِٹ چھُ اکھ اختیار، مگر پرٛتھ کٲنٛسہِ چھُنہ اَمہِچ ضرورت آسان؛ وُچھِو زِ پرٛتھ جایِہ حقیقتن منز کٔم مدد دِوان چھِ۔

## دورٕ دوران مےٚ کیا گژِھ نظرِ منز تھاوُن؟

اگر ممکن آسِہ، تیٚلہِ کٔرِو اکھ کھوتہٕ زیاده گرن ہُند دورہ۔ گۄڈنِچہِ دورٕ خٲطرٕ ٹھہرٲوِو وقت، پتہٕ پٕرژھِو زِ کیا تُہؠ ہیکِو بییہِ کُنہِ الگ وقتس یا کھینَس وقتس دورہ کٔرِتھ۔ دوٚیِم دورس منز ہیکِہ معمولچ زیادہ تصویر سامنہ یِتھ تہٕ تُہؠ ہیکِو باقی عملہ سٍتۍ تہِ مِلِتھ۔

وُچھِو زِ عملہ کِتھ کٔنۍ چھُ باشندن سٍتۍ واتان۔ کیا تِم چھا لُکن عزت سان مخاطب کران، جواب دِنہ خٲطرٕ وقت دِوان تہٕ عام فٲصلن منز تِمن شٲمِل کران؟ کیا باشندٕ چھا آرام دہ محسوس کران تہٕ تِہِندین شوقن تہٕ صلاحیتن مُطٲبق کینٛہہ کرنہ خٲطرٕ چھا؟ پٕرژھِو زِ کیا تُہؠ ہیکِو سرگرمین ہٕنز وقت سارنی، کھین جاے، نبر جاے تہٕ ذاتی جاے وُچھِتھ۔

تِمن تفصیلن پیٹھ دِیِو توجہ یِم یَتھ شخصس خٲطرٕ اہم چھِ: آسان سمجھنہ یِنہ وٲلؠ نشان، باتھ رومس تام واتنٕک صاف وَتھ، مانوس چیزٕ، پرسکون جاے، زبان تہٕ ثقافتی مدد، تہٕ پننین عزیزن سٍتۍ مِلنٕک موقع۔ صرف خوبصورت لابی چھَنہ یہِ ظٲہر کران زِ روزمرٕ دیکھ بھال کِتھ کٔنۍ چھِ چلان۔

اگر یہِ تُہنزٕ سوالات یاد تھاونس تہٕ پتہٕ اثراتن ہُند موازنہ کرنس منز مدد کرِ، تیٚلہِ نِیِو پنُن کانٛہہ بھروسہ مند دوست یا رشتہ دار سٍتۍ۔ تُہؠ ہیکِو [Alzheimer's Association ہُند چیک لسٹ](https://www.alz.org/getmedia/2f238875-3f5d-447e-adcc-dd9fc07f828b/alzheimers-dementia-choosing-residential-care-ts.pdf) استعمال کٔرِتھ یہِ نوٹ کٔرِتھ زِ تُہؠ کیا وُچھِو۔ پننین نوٹن سمجھِو کتھ باتھ خٲطرٕ اشارٕ، نہ زِ کانٛہہ تِیُتھ سکور یُس تُہندِ خٲطرٕ گھر چُنِ۔

## دیکھ بھال ہُند موازنہ کرنہ وقتہٕ کٔمہِ سوال چھِ مددگار؟

پٕرژھِو زِ دۄہس کینٛہن مختلف وقتن منز عملہ کِتھ کٔنۍ چھُ کام کران، عملہ کِتھ کٔنۍ چھُ شخص سٕند معمولات زانان تہٕ تِہٕنز ضرورتہٕ بدلنس پیٹھ کیا گژھان چھُ۔ پٕرژھِو زِ دیکھ بھال منصوبن ہٕنز نظرِ ثانی کُس کران چھُ تہٕ خاندان کِتھ کٔنۍ ہیکِہ کانٛہہ فِکر ظاہر کٔرِتھ۔ اگر شخصس کھینَس، ذاتی دیکھ بھالس یا رابطس منز مدد چھِ ضرورت، تیٚلہِ پٕرژھِو زِ سۄ مدد عملن منز کِژاہ نظر یِوان چھِ۔

خاندانی ملاقاتن، نبر نیرنہ، ڈاکٹری تقررن، سرگرمین تہٕ شخص سٕنٛزِ مانوس چیزٕ اَننٕچ صلاحیت مُتعلق پٕرژھِو۔ اگر کُنہِ گھرَس منز ڈیمنشیا خٲطرٕ خاص پروگرام چھُ، تیٚلہِ پٕرژھِو زِ اتھ منز کٔم تربیت، سرگرمیہٕ تہٕ روزمرٕ مدد شٲمِل چھِ۔ صرف پروگرام ناوَس پیٹھ بھروسہ کرنہ بجایِہ پٕرژھِو زِ باشندٕ کِتھ کٔنۍ چھِ فٲصلن منز شٲمِل یِوان کرنہ۔

فیسن ہٕنز تفصیلی بیان منگِو، زِ اتھ منز کیا شٲمِل چھُ تہٕ خرچ کِتھ کٔنۍ ہیکَن بدلِتھ۔ معاہدہ غور سان پٔرِو تہٕ اگر مالی یا قانونی شرائط صاف آسَن نہ، تیٚلہِ نِیِو اہل مقامی مشورہ۔ یہِ نہ سمجھِو زِ کانٛہہ خدمت دستیاب چھِ یا کورڈ چھِ صرف اَمی مۄکھہ زِ بییہِ کُنہِ ریاست یا ملک منز تِژ معلومات آیہِ دِنہ۔

اگر شخص سٕنٛز شناخت، رشتہٕ یا رازداری خٲطرٕ اضافی توجہ چھِ ضرورت، تیٚلہِ وُچھِو سون [ڈیمنشیا سانٛ زندگی گزاران LGBTQ+ شخصس مدد کرنٕچ رہنمائی](/caregiving/supporting-lgbtq-person-with-dementia/)۔ اتھ منز پسندیدہ ناون، ملاقات کرن والین، ذاتی دیکھ بھال تہٕ فراہم کرن والین ہٕنٛزن پالیسین مُتعلق سوال تہِ شٲمِل چھِ۔

## تبدیلی منز بہٕ کِتھ کٔنۍ ہیکہٕ مدد کٔرِتھ؟

شخصس کٔرِو فٲصلن منز تیوٗت شٲمِل یِیوٗت سُہ چاہِ۔ اگر دستیاب آسِہ تہٕ تِہِندِ خٲطرٕ ٹھیک آسِہ، تیٚلہِ گھرُک اکھ دورہ، تَتہِ اکھ کھین یا اکھ لۄکُٹ آرامُک قیام ہیکِہ کینٛہن لُکن خٲطرٕ تَتہِ سٍتۍ مانوس گژھنہ منز مددگار ٲسِتھ۔ شخص ہیکِہ کمرٕچ کانٛہہ چیز، کِراہ پالَو یا کانٛہہ معمول چُنُن پسند کٔرِتھ یُس سُہ سٍتۍ نِتھ ہیکِہ۔

شخص سٕنٛزِ منظوری سان کٔرِو عملہ سٍتۍ مفید معلومات شیئر۔ اتھ منز ہیکِہ تِہُند پسندیدہ ناو، رابطہ کرنُک طریقہ، روزمرٕ معمولات، تِم کھین یِم تِمن پسند چھِ، تِمہٕ سرگرمیہٕ یِم تِمن مزہ دِوان چھِ، تہٕ سُہ کیا یُس تِمن سکون محسوس کرناوان چھُ شٲمِل ٲسِتھ۔ معلومات تھٲوِو مختصر تہٕ دیکھ بھالس پیٹھ مرکوز۔ شخص گژِھ پننِس بارس منز گژھن والین کتھ باتھن ہُند حصّہ روزُن۔

اگر گھر اجازت دِیِہ، تیٚلہِ نِیِو مانوس چیزٕ سٍتۍ۔ عملس پٕرژھِو زِ شخص سُند کمرہ کِتھ کٔنۍ ہیکِہ مانوس بٔنِتھ، مگر تِیُتھ تہِ نہ زِ اتھ منز اَتھ پھیرِتھ گژُھن مشکل گژِھ۔ اکھ تصویر، پسندیدٕ کمبل یا مانوس گھڑی ہیکِہ پوٗرٕ گرِ ہُند ماحول بناونہ کھوتہٕ زیادہ اہم ٲسِتھ۔

## تبدیلی پتہٕ بہٕ کِتھ کٔنۍ ہیکہٕ شٲمِل روزِتھ؟

عملہ سٍتۍ میل جول کٔرِتھ زٲنِو زِ نوِ معلومات کِتھ کٔنۍ چھِ شیئر یِوان کرنہ تہٕ کَس سٍتۍ گژِھ رابطہ کرُن۔ شخصس پٕرژھِو زِ سُہ ملاقاتن تہٕ سرگرمین بارس منز کِژاہ محسوس کران چھُ۔ کیہنہ لُکھ ہیکَن واریاہ چھوٹی ملاقات پسند کٔرِتھ؛ باقی ہیکَن سکون سان رۄزِنہ خٲطرٕ کینٛہہ وقت یژِھتھ۔ وُچھِو زِ تِہِندِ خٲطرٕ کیا چھُ آرام دہ باسان تہٕ مِلِتھ کٔرِو تبدیلی۔

دیکھ بھال ضرورتہٕ تہٕ ترجیحہٕ ہیکَن بدلِتھ۔ شخصس تہٕ عملس سٍتۍ کٔرِو دیکھ بھال منصوبٕچ نظرِ ثانی، اگر کانٛہہ چیز صاف آسِہ نہ تیٚلہِ پٕرژھِو، تہٕ تِژ نوِ معلومات شیئر کٔرِو یِم مددگار ہیکَن ٲسِتھ۔ اگر شخصس اچانک پریشانی یا واریاہ زیادٕ بے حالی محسوس گژِھ، تیٚلہِ کٔرِو فوراً دیکھ بھال ٹیمس اطلاع؛ [NHS](https://www.nhs.uk/symptoms/confusion/) مُطٲبق چھِ اچانک پریشانی خٲطرٕ فوری طبی توجہ ضرورت۔

گھر چُنُن چھُ اکھ سنجیدٕ فٲصلہ، تہٕ گۄڈنِک انتخاب ہیکِہ ہمیشہ خٲطرٕ بہترین نہ روزِتھ۔ معلومات جمع کٔرِو، شخصس شٲمِل کٔرِو، روزمرٕ دیکھ بھال مُتعلق پٕرژھِو، تہٕ تبدیلی پتہٕ تہِ تھٲوِو رابطہ جاری۔
