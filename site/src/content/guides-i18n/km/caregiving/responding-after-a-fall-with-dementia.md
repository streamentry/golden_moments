## តើខ្ញុំគួរធ្វើអ្វីភ្លាមៗបន្ទាប់ពីដួល?

ពិនិត្យថាតើអ្នកនោះនៅភ្ញាក់ មានរបួស ឬមិនស្រួលខ្លួនឬទេ ហើយអាចឆ្លើយតបដូចធម្មតាឬទេ។ កុំប្រញាប់ឲ្យពួកគេក្រោកឈរ ឬព្យាយាមលើកពួកគេដោយខ្លួនឯង។ [សេវាសុខាភិបាលអង់គ្លេស NHS និយាយថាត្រូវការជំនួយសង្គ្រោះបន្ទាន់ បើអ្នកណាម្នាក់អាចបានរងរបួសក្បាល ខ្នង កអាវ ឬត្រគាក មានរបួសធ្ងន់ធ្ងរផ្សេងទៀត ឬមិនអាចក្រោកបាន](https://www.nhs.uk/conditions/falls/)។

- **ហៅសេវាសង្គ្រោះបន្ទាន់ក្នុងតំបន់** ប្រសិនបើសញ្ញាធ្ងន់ធ្ងរមួយណាក្នុងចំណោមនោះអាចមាន។ ធ្វើតាមការណែនាំរបស់អ្នកទទួលការហៅ។
- **ផ្តល់ឲ្យអ្នកនោះមានផាសុកភាព និងកម្តៅ** ខណៈរង់ចាំជំនួយ។ កុំផ្លាស់ទីពួកគេ ប្រសិនបើអ្នកសង្ស័យថាមានរបួសធ្ងន់ធ្ងរ ឬមិនប្រាកដថាធ្វើដូច្នេះមានសុវត្ថិភាព។
- **នៅជាមួយពួកគេ ប្រសិនបើអ្នកអាច។** ប្រើពាក្យខ្លី និងស្ងប់។ កុំសួរសំណួរដេញដោលអំពីអ្វីដែលបានកើតឡើង ឬទទូចឲ្យពួកគេក្រោកឈរ។
- **ប្រសិនបើអ្នកនោះឈឺចាប់ រងរបួស ឬមិនស្រួលខ្លួន ប៉ុន្តែគ្មានគ្រោះថ្នាក់ភ្លាមៗ** សូមទាក់ទងការណែនាំវេជ្ជសាស្ត្របន្ទាន់ក្នុងតំបន់ ឬអ្នកជំនាញសុខភាពរបស់ពួកគេ ដើម្បីសុំការណែនាំ។

កុំព្យាយាមបកឬលើកអ្នកនោះតែម្នាក់ឯង។ ប្រសិនបើមានផែនការថែទាំសម្រាប់ការដួល សូមអនុវត្តតាម តែនៅពេលវាស្របនឹងអ្វីដែលកំពុងកើតឡើងឥឡូវនេះ ប៉ុណ្ណោះ។ ប្រសិនបើអ្នកនោះអាចរងរបួសធ្ងន់ធ្ងរ ត្រូវស្វែងរកជំនួយសង្គ្រោះបន្ទាន់។

## តើធ្វើដូចម្ដេច ប្រសិនបើអ្នកនោះហាក់ដូចជាអាចក្រោកបាន?

កុំប្រញាប់អ្នកនោះ។ សួរថាតើពួកគេមានអារម្មណ៍យ៉ាងណា ហើយតើពួកគេចង់បានជំនួយឬទេ។ ប្រសិនបើពួកគេហាក់ដូចជាមិនរងរបួស ហើយអាចក្រោកបានដោយប្រើការគាំទ្រធម្មតារបស់ពួកគេ ដោយមិនត្រូវលើក ផ្តល់ពេលវេលាឲ្យពួកគេ ហើយនៅជិត។ ឈប់ប្រសិនបើពួកគេឈឺ វិលមុខ ឬមិនអាចធ្វើបាន។ ប្រសិនបើអ្នកមិនប្រាកដ សូមសួរអ្នកជំនាញសុខភាព ជាជាងព្យាយាមផ្លាស់ទីដែលអ្នកមិនត្រូវបានបង្ហាញពីរបៀបធ្វើ។

[NHS ណែនាំឲ្យយកពេលវេលា និងសម្រាកបន្ទាប់ពីក្រោកឡើង](https://www.nhs.uk/conditions/falls/)។ ការណែនាំជួយខ្លួនឯងរបស់ពួកគេសម្រាប់អ្នកដែលមានអារម្មណ៍ថាអាចក្រោកបាន ហើយមិនមែនជាហេតុផលសម្រាប់ទាញ ឬលើកអ្នកដែលដួលឡើយ។ សមត្ថភាពចលនា តុល្យភាព និងតម្រូវការសុខភាពរបស់មនុស្សម្នាក់ៗមានភាពខុសគ្នា។

## តើខ្ញុំគួរទាក់ទងគ្រូពេទ្យទោះបីមើលមិនឃើញរបួសឬទេ?

គួរ។ [សមាគមអាល់ស៊ីមឺរណែនាំឲ្យនិយាយជាមួយគ្រូពេទ្យទូទៅ (GP) របស់អ្នកនោះបន្ទាប់ពីដួល ទោះបីមិនគិតថាពួកគេរងរបួសក៏ដោយ](https://dementiaconnect.alzheimers.org.uk/blog/why-person-dementia-more-likely-have-fall-and-what-can-you-do-help)។ ប្រសិនបើអ្នកបានកត់សម្គាល់ការប៉ះក្បាល ការឈឺចាប់ ភាពច្របូកច្របល់ថ្មី ភាពទន់ខ្សោយ ការវិលមុខ ឬការផ្លាស់ប្តូរពីអាកប្បកិរិយាធម្មតារបស់អ្នកនោះ សូមរាយការណ៍ពីវានៅពេលអ្នកហៅ។ កុំប្រើបញ្ជីពិនិត្យតាមគេហទំព័រ ដើម្បីសម្រេចថាការវាយតម្លៃវេជ្ជសាស្ត្រមិនចាំបាច់។

ប្រសិនបើមានការប៉ះក្បាល សូមអនុវត្តតាមដំបូន្មានពីគ្រូពេទ្យ ឬសេវាថែទាំបន្ទាន់ក្នុងតំបន់។ [ការណែនាំអំពីការប៉ះក្បាលរបស់ NHS](https://www.nhs.uk/conditions/head-injury-and-concussion/) ពិពណ៌នារោគសញ្ញាដែលត្រូវការការយកចិត្តទុកដាក់បន្ទាន់។ ប្រសិនបើអ្នកនោះស្រាប់តែច្របូកច្របល់ ងងុយគេងខ្លាំង ក្អួត ឬហាក់ដូចជាមិនស្រួលខ្លួនយ៉ាងធ្ងន់ធ្ងរ សូមស្វែងរកជំនួយបន្ទាន់ ជាជាងរង់ចាំការណាត់ជួបធម្មតា។

## តើព័ត៌មានអ្វីអាចជួយអ្នកជំនាញសុខភាព?

បន្ទាប់ពីតម្រូវការបន្ទាន់ត្រូវបានដោះស្រាយ សូមសរសេរការសង្កេតខ្លះៗ ដើម្បីចែករំលែកដោយមានការយល់ព្រមពីអ្នកនោះ៖

- ពេលវេលា និងកន្លែងដែលការដួលបានកើតឡើង បើដឹង;
- អ្វីដែលអ្នកនោះកំពុងធ្វើមុនពេលដួល;
- ថាតើមានអ្នកឃើញការដួលឬទេ ហើយថាតើអ្នកនោះប្រហែលជាបានប៉ះក្បាលឬទេ;
- ការឈឺចាប់ របួស ឬការផ្លាស់ប្តូរអ្វីដែលអ្នកបានកត់សម្គាល់;
- ការផ្លាស់ប្តូរថ្មីៗណាមួយក្នុងសុខភាព ចលនា ទម្លាប់ ឬថ្នាំ ដែលអ្នកដឹង។

ពិពណ៌នាអ្វីដែលអ្នកបានឃើញ ជាជាងទាយថាហេតុអ្វីបានជាអ្នកនោះដួល។ កំណត់ចំណាំអាចជួយការសន្ទនា ប៉ុន្តែវាមិនវិនិច្ឆ័យមូលហេតុទេ។ ប្រសិនបើអាចមានរបួសធ្ងន់ធ្ងរ ឬការផ្លាស់ប្តូរភ្លាមៗ សូមទទួលជំនួយជាមុនសិន ជាជាងបំពេញកំណត់ត្រា។

## តើយើងគួរសួរអ្វីក្នុងការតាមដាន?

អ្នកជំនាញវេជ្ជសាស្ត្រអាចជួយសម្រេចថាតើអ្នកនោះត្រូវការការវាយតម្លៃឬទេ ហើយអ្វីដែលត្រូវមើលបន្ទាប់ពីនោះ។ [NHS បញ្ជាក់ថាការវាយតម្លៃការដួលអាចពិនិត្យកម្លាំង ការមើលឃើញ ការស្តាប់ ថ្នាំ និងការកែតម្រូវផ្ទះ](https://www.nhs.uk/conditions/falls/)។ អ្នកអាចសួរថា៖

- តើអ្នកនោះត្រូវការការពិនិត្យបន្ទាប់ពីដួលនេះឬទេ ទោះបីមិនឃើញមានរបួសច្បាស់ក៏ដោយ?
- ការផ្លាស់ប្តូរ ឬរោគសញ្ញាណាដែលគួរតែជំរុញឲ្យស្វែងរកជំនួយបន្ទាន់?
- តើតុល្យភាព ការមើលឃើញ ការស្តាប់ ការឈឺចាប់ ឬថ្នាំគួរត្រូវបានពិនិត្យឬទេ?
- តើសេវាដួល ឬអ្នកព្យាបាលការងារ អាចជួយវាយតម្លៃចលនា ឬឧបករណ៍ក្នុងផ្ទះបានទេ?
- តើអ្នកនោះ ឬគ្រួសារគួរទាក់ទងនរណា ប្រសិនបើមានការដួលម្តងទៀត?

ប្រាប់អ្នកជំនាញអំពីអ្វីដែលអ្នកដឹង និងអ្វីដែលអ្នកមិនដឹង។ ប្រសិនបើមិនមានអ្នកឃើញការដួល សូមប្រាប់ឲ្យបានច្បាស់ ជាជាងទាយថាវាកើតឡើងដោយរបៀបណា។ កុំឈប់ ឬផ្លាស់ប្តូរថ្នាំ ទិញឧបករណ៍ដើរ ឬដំឡើងឧបករណ៍ ដោយផ្អែកតែលើគេហទំព័រ។ ការគាំទ្រដែលត្រឹមត្រូវអាស្រ័យលើសុខភាព ចលនា ផ្ទះ និងបំណងប្រាថ្នារបស់អ្នកនោះ។

## តើយើងអាចគិតពីការដួលម្តងទៀតដូចម្ដេច?

ពេលបញ្ហាវេជ្ជសាស្ត្រភ្លាមៗត្រូវបានដោះស្រាយ សូមពិចារណាអំពីអ្វីដែលកំពុងកើតឡើងនៅកន្លែង និងទម្លាប់។ ផ្លូវដែលមានរបស់ដាក់ច្រើន ការផ្លាស់ប្តូរពន្លឺ ឬផ្លូវដែលមិនធ្លាប់ស្គាល់ អាចពាក់ព័ន្ធ ប៉ុន្តែកុំសន្មតថាគ្រោះថ្នាក់តែមួយបានបណ្តាលឲ្យដួល។ សួរអ្នកនោះថាពួកគេបានកត់សម្គាល់អ្វី និងអ្វីដែលនឹងធ្វើឲ្យកន្លែងងាយប្រើជាង។ ប្រសិនបើអ្នកនោះចាំ ឬពន្យល់ការដួលមិនបាន សូមចែករំលែកអ្វីដែលអ្នកបានសង្កេតឃើញ ដោយមិនសង្កត់ឲ្យពួកគេផ្តល់ព័ត៌មានលម្អិត។ កត់ត្រាពេលវេលា ទីតាំង ស្បែកជើង ពន្លឺ សកម្មភាព និងការផ្លាស់ប្តូរណាមួយពីចលនា ឬអាកប្បកិរិយាធម្មតារបស់ពួកគេ។ កំណត់ចំណាំខ្លីអាចជួយអ្នកជំនាញស្វែងរកលំនាំ ប៉ុន្តែវាមិនកំណត់អត្តសញ្ញាណមូលហេតុទេ។

[មគ្គុទ្ទេសក៍សុវត្ថិភាពផ្ទះ](/caregiving/home-safety/) របស់យើង គ្របដណ្តប់ការពិនិត្យតាមបន្ទប់នីមួយៗ។ ប្រសិនបើការដួលកើតឡើងពេលងូតទឹក សូមមើល[មេរៀនអំពីការងូតទឹក](/caregiving/bathing-with-dementia/)។ គ្រូពេទ្យ អ្នកព្យាបាលការងារ ឬសេវាដួលក្នុងតំបន់ អាចវាយតម្លៃកង្វល់បុគ្គល និងណែនាំការផ្លាស់ប្តូរសមស្រប។ កុំដំឡើងឧបករណ៍ ឬផ្លាស់ប្តូរថ្នាំដែលត្រូវបានចេញវេជ្ជបញ្ជា ដោយផ្អែកតែលើទំព័រនេះ។

ការដួលមួយមិនបង្ហាញដោយខ្លួនឯងថាអ្នកនោះអាចធ្វើអ្វី ឬមិនអាចធ្វើអ្វីឡើយ។ រក្សាពួកគេឲ្យចូលរួមក្នុងការសម្រេចចិត្តអំពីការគាំទ្រ ហើយអនុវត្តតាមដំបូន្មានរបស់ក្រុមថែទាំក្នុងតំបន់។
