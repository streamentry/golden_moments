## តើខ្ញុំគួរចាប់ផ្តើមការសន្ទនាពេលណា?

ចាប់ផ្តើមមុនពេលហេតុការណ៍បើកបរមួយបង្ខំឲ្យមានការសម្រេចចិត្តបន្ទាន់ ប្រសិនបើអាច។ ជ្រើសរើសពេលស្ងប់ស្ងាត់ នៅពេលដែលអ្នកដែលមានជំងឺមិនកំពុងត្រៀមទៅណាមួយ ឬត្រឡប់ពីដំណើរដែលមានភាពតានតឹង។ នេះអាចជាការសន្ទនាដំបូងក្នុងចំណោមការសន្ទនាជាច្រើនដង មិនមែនជាការប្រកាសតែមួយដងទេ។

សួរថាការបើកបរមានន័យអ្វីសម្រាប់អ្នកដែលមានជំងឺ។ វាអាចជាការទៅសួរសុខទុក្ខមិត្តភក្តិ ទិញរបស់ប្រើប្រាស់ ជួយគ្រួសារ ធ្វើការងារ ឬមានអារម្មណ៍ឯករាជ្យ។ ការទទួលស្គាល់អត្ថន័យនោះមិនបានដោះស្រាយបញ្ហាសុវត្ថិភាពទេ។ វាបង្ហាញថាអ្នកយល់ពីអ្វីដែលអាចបាត់បង់ និងហេតុអ្វីការរៀបចំមានសារៈសំខាន់។

រួមបញ្ចូលអ្នកដែលមានជំងឺក្នុងការជ្រើសរើសការធ្វើដំណើរនាពេលអនាគត ខណៈពួកគេចង់ចូលរួម។ ពួកគេអាចចូលចិត្តឲ្យមនុស្សជាក់លាក់ម្នាក់បើកបរ ផ្លូវធម្មតា ឬពេលវេលាជាក់លាក់នៃថ្ងៃ។ សូមកត់ត្រាជម្រើសដែលពួកគេចូលចិត្ត ហើយសាកល្បងពួកវាមុនពេលថ្ងៃដែលមានភាពតានតឹង ប្រសិនបើពួកគេយល់ព្រម។

## តើខ្ញុំអាចលើកការព្រួយបារម្ភដោយមិនធ្វើឲ្យវាក្លាយជាការជជែកតវ៉ាដោយរបៀបណា?

និយាយអំពីព្រឹត្តិការណ៍ជាក់លាក់ ជាជាងធ្វើការវិនិច្ឆ័យទូលំទូលាយ។ ឧទាហរណ៍ ពិពណ៌នាពីការវង្វេងលើផ្លូវធម្មតា ការមិនឃើញសញ្ញាឈប់ ឬការប៉ះធ្ងន់លើរថយន្តថ្មីៗ។ និយាយអ្វីដែលអ្នកបានឃើញ ហើយសួរថាអ្នកដែលមានជំងឺចងចាំ ឬមានអារម្មណ៍អ្វី។ ជៀសវាងការហៅពួកគេថាជាអ្នកបើកបរមិនល្អ ឬសាកល្បងការចងចាំរបស់ពួកគេដោយសំណួរ។

ស្តាប់ទស្សនៈរបស់អ្នកដែលមានជំងឺ។ ពួកគេអាចបានកត់សម្គាល់ការច្របូកច្របល់ ពន្លឺចាំងភ្នែក ភាពអស់កម្លាំង ឬការលំបាកផ្សេងទៀតដែរ។ ឬពួកគេអាចមិនយល់ស្របនឹងការព្រួយបារម្ភរបស់អ្នក។ កុំយកការសន្ទនាតែមួយដងជាភស្តុតាងថាពួកគេយល់ ឬមិនយល់ពីហានិភ័យឡើយ។ រក្សាសំឡេងឲ្យគោរព ហើយត្រឡប់ទៅប្រធានបទនេះនៅពេលក្រោយ ប្រសិនបើម្នាក់ណាមួយមានការខឹង។

អ្នកឯកទេសអូស្ត្រាលី Dementia Australia ណែនាំឲ្យមានការសន្ទនាខ្លីៗ ម្តងហើយម្តងទៀត ហើយផ្តោតលើចំណុចខ្លាំងរបស់អ្នកដែលមានជំងឺ និងជម្រើសធ្វើដំណើរជាវិជ្ជមាន។ ការលាក់គ្រាប់ចុចឬបិទរថយន្តដោយសម្ងាត់ អាចមានអារម្មណ៍ខ្មាំងសត្រូវក្នុងដំណាក់កាលរៀបចំដំបូង។ ប្រសិនបើមានហានិភ័យសុវត្ថិភាពធ្ងន់ធ្ងរភ្លាមៗ សូមធ្វើជំហានដើម្បីជៀសវាងការធ្វើដំណើរហានិភ័យបន្ទាប់ និងស្វែងរកការណែនាំវិជ្ជាជីវៈភ្លាមៗ។

## តើព័ត៌មានអ្វីអាចជួយការពិភាក្សាជាមួយអ្នកជំនាញ?

សរសេរព្រឹត្តិការណ៍ជាក់ស្តែង និងពេលដែលវាកើតឡើង។ ឧទាហរណ៍អាចរួមមានការច្របូកច្របល់នៅចំណុចប្រសព្វ ការបាត់ផ្លូវទៅកន្លែងធម្មតា ការមិនឃើញសញ្ញា ឬការខូចខាតថ្មីលើរថយន្ត។ រួមបញ្ចូលអ្វីដែលអ្នកដែលមានជំងឺបាននិយាយអំពីការធ្វើដំណើរ និងការផ្លាស់ប្តូរសុខភាព ឬឱសថណាមួយដែលពួកគេចង់ពិភាក្សា។

រក្សាកំណត់ចំណាំឲ្យត្រូវនឹងការពិត និងឯកជន។ កុំថតសំឡេងដោយសម្ងាត់ តាមដាន ឬសាកល្បងអ្នកដែលមានជំងឺ។ ប្រសិនបើអ្នកដែលមានជំងឺយល់ព្រម សមាជិកគ្រួសារដែលធ្លាប់អង្គុយក្នុងរថយន្តអាចចែករំលែកអ្វីដែលពួកគេបានសង្កេតឃើញ។ ការពិពណ៌នាច្បាស់លាស់ផ្តល់ឲ្យគ្រូពេទ្យ ឬអ្នកជំនាញនូវអ្វីជាក់លាក់ដើម្បីពិភាក្សា ប៉ុន្តែវាមិនវិនិច្ឆ័យមូលហេតុ ឬកំណត់សមត្ថភាពបើកបរឡើយ។

សញ្ញាព្រមានក្នុង[ការណែនាំរបស់សមាគម Alzheimer's Association](https://www.alz.org/help-support/caregiving/safety/dementia-driving) គឺជាហេតុផលដើម្បីស្វែងរកការណែនាំវិជ្ជាជីវៈ។ ពួកវាមិនជំនួសការវាយតម្លៃបុគ្គលទេ។ សួរគ្រូពេទ្យរបស់អ្នកដែលមានជំងឺអំពីការណែនាំទៅអ្នកព្យាបាលការងារ ឬអ្នកជំនាញស្តារការបើកបរដែលមានសមត្ថភាពផ្សេងទៀត។ ពិនិត្យច្បាប់អាជ្ញាប័ណ្ណក្នុងតំបន់ជាមួយអាជ្ញាធរអាជ្ញាប័ណ្ណរដ្ឋ ឬតំបន់។

## តើយើងអាចធ្វើផែនការធ្វើដំណើរជាមួយគ្នាដោយរបៀបណា?

ចាប់ផ្តើមជាមួយដំណើរដែលអ្នកដែលមានជំងឺចង់រក្សា៖ ការទៅសួរសុខទុក្ខមិត្តភក្តិ ការទិញរបស់ ការណាត់ជួប ការគោរពប្រណិប័តន៍ ឬសកម្មភាពធម្មតា។ សម្រាប់ដំណើរនីមួយៗ សូមពិភាក្សាអំពីអ្នកណាអាចបើកបរ ជម្រើសដឹកជញ្ជូនណាដែលមានអារម្មណ៍អាចទទួលយកបាន និងការគាំទ្របែបណាដែលនឹងធ្វើឲ្យដំណើរនោះដំណើរការ។

ជម្រើសអាចរួមមានការជិះពីគ្រួសារ ឬមិត្តភក្តិ តាក់ស៊ី ឬសេវាចែករំលែកការធ្វើដំណើរ ដឹកជញ្ជូនសាធារណៈ ការដឹកជញ្ជូនក្នុងសហគមន៍ ឬការដឹកជញ្ជូនទំនិញសម្រាប់ការទិញខ្លះ។ សេវា ថ្លៃ និងភាពងាយស្រួលខុសគ្នាតាមទីតាំង។ ពិនិត្យអ្វីដែលមាននៅក្នុងតំបន់ជាជាងសន្មតថាជម្រើសមួយនឹងសមនឹងមនុស្សគ្រប់គ្នា។

សាកល្បងផ្លូវ ឬសេវាជាមួយគ្នាមុនពេលវាក្លាយជាជម្រើសតែមួយ។ សួរថាតើអ្នកដែលមានជំងឺចង់មានមិត្តរួមដំណើរ ឬចូលចិត្តធ្វើដំណើរដោយខ្លួនឯងឬអត់។ [មគ្គុទ្ទេសក៍ការរៀបចំដំណើររបស់យើង](/caregiving/traveling-with-dementia/) មានគំនិតសម្រាប់ធ្វើឲ្យដំណើរនៅតែធម្មតា និងទុកកន្លែងសម្រាប់ផ្លាស់ប្តូរផែនការ។

## តើធ្វើដូចម្តេចប្រសិនបើខ្ញុំគិតថាដំណើរបន្ទាប់អាចមិនមានសុវត្ថិភាព?

សម្រាប់ការព្រួយបារម្ភធ្ងន់ធ្ងរផ្សេងទៀត សូមទាក់ទងគ្រូពេទ្យរបស់អ្នកដែលមានជំងឺ និងអាជ្ញាធរអាជ្ញាប័ណ្ណក្នុងតំបន់ ដើម្បីដឹងជំហានដែលអនុវត្តនៅកន្លែងដែលពួកគេរស់នៅ។ កុំពឹងផ្អែកលើសន្យាគ្រួសារ ឬទំព័រនេះដើម្បីសម្រេចតម្រូវការច្បាប់។ ប្រសិនបើអាច សូមពន្យល់ផែនការភ្លាមៗដល់អ្នកដែលមានជំងឺ ហើយរក្សាពួកគេឲ្យចូលរួមក្នុងអ្វីដែលនឹងកើតឡើងបន្ទាប់។

## តើខ្ញុំអាចគាំទ្រអ្នកដែលមានជំងឺដោយរបៀបណា ប្រសិនបើការបើកបរបានបញ្ចប់?

ការឈប់បើកបរអាចមានអារម្មណ៍ដូចជាការបាត់បង់ដ៏ធំ។ ទទួលស្គាល់វាដោយត្រង់ៗ។ បន្តអញ្ជើញអ្នកដែលមានជំងឺទៅកន្លែង និងសកម្មភាពដែលសំខាន់ចំពោះពួកគេ។ ផ្តល់ជម្រើសអំពីការជិះ ពេលវេលា និងមិត្តរួមដំណើរ ជាជាងចាត់ទុកការធ្វើដំណើរជាកិច្ចការដែលត្រូវគ្រប់គ្រងជំនួសពួកគេ។

ចែករំលែកកាតព្វកិច្ចបើកបរជាមួយញាតិ ឬមិត្តភក្តិ ប្រសិនបើអ្នកដែលមានជំងឺយល់ព្រម។ សួរទីភ្នាក់ងារសហគមន៍អំពីការដឹកជញ្ជូនសម្រាប់មនុស្សពេញវ័យ ឬមនុស្សដែលមានពិការភាព។ ពិនិត្យថ្លៃ កាលវិភាគ ច្បាប់ការទទួល និងថាតើអនុញ្ញាតឲ្យមានមិត្តរួមដំណើរឬអត់។ ព័ត៌មានលម្អិតតូចៗអាចកំណត់ថាតើជម្រើសមួយមានប្រយោជន៍ឬអត់។

[មគ្គុទ្ទេសក៍ថែទាំខ្លួនឯងសម្រាប់អ្នកថែទាំ](/caregiving/caregiver-self-care/) គ្របដណ្តប់លើការសុំឲ្យអ្នកដទៃចែករំលែកការគាំទ្រ។ អ្នកដែលមានជំងឺគួរមានការសម្រេចចិត្តច្រើនបំផុតតាមដែលអាច លើរបៀបដែលពួកគេនៅតែភ្ជាប់ទំនាក់ទំនងជាមួយជីវិតប្រចាំថ្ងៃ។

គោលបំណងគឺការពារសុវត្ថិភាព ខណៈដែលគោរពអត្ថន័យរបស់ការបើកបរចំពោះអ្នកដែលមានជំងឺ។ ចាប់ផ្តើមការសន្ទនាឆាប់ៗ ស្តាប់ ទទួលការណែនាំវិជ្ជាជីវៈ ហើយសាងផែនការធ្វើដំណើរជាមួយគ្នា។
