## តើជំហានដំបូងដ៏មានប្រយោជន៍មួយជាអ្វី?

ជ្រើសរើសសំណួរមួយដែលនឹងធ្វើឲ្យសប្តាហ៍នេះងាយស្រួលជាងមុន។ វាអាចជាការស្វែងរកថានរណានឹងសម្របសម្រួលការតាមដានសុខភាពរបស់អ្នកដែលមានជំងឺ ការសួរកន្លែងដែលអាចទទួលបានព័ត៌មានដែលអាចទុកចិត្តបានអំពីការធ្វើរោគវិនិច្ឆ័យ ឬការកំណត់អ្នកដែលខ្លួនទុកចិត្តឲ្យនិយាយជាមួយ។ អ្នកដែលមានជំងឺអាចចង់ដឹកនាំខ្លួនឯង ស្នើឲ្យនរណាម្នាក់ជួយ ឬរង់ចាំមុនពេលសម្រេចចិត្ត។

សម្រាប់ការណាត់ជួបពេលក្រោយ សូមសរសេរសំណួរមួយចំនួនជាមួយគ្នា ប្រសិនបើអ្នកដែលមានជំងឺចង់។ អ្នកអាចសួរថាការធ្វើរោគវិនិច្ឆ័យមានន័យយ៉ាងណា អ្វីខ្លះនៅតែមិនទាន់ច្បាស់ ពេលណាត្រូវតាមដាន និងត្រូវទាក់ទងនរណា ប្រសិនបើមានការព្រួយបារម្ភថ្មីកើតឡើង។ សូមសុំឲ្យគ្រូពេទ្យពន្យល់ពាក្យ ឬការណែនាំណាមួយដែលមិនច្បាស់។ [មគ្គុទ្ទេសក៍ការទៅជួបគ្រូពេទ្យ](/caregiving/doctor-visit-with-dementia/) របស់យើងមានបញ្ជីរៀបចំខ្លីមួយ។

អ្នកដែលមានជំងឺមិនចាំបាច់ដឹងអ្វីគ្រប់យ៉ាងអំពីជំងឺភ្លេចភ្លាំងក្នុងពេលតែមួយទេ។ ជ្រើសរើសព័ត៌មានដែលអាចទុកចិត្តបានអំពីការធ្វើរោគវិនិច្ឆ័យដែលខ្លួនបានទទួល ហើយកត់ត្រាសំណួរណាដែលកើតឡើងនៅពេលក្រោយ។ សូមប្រយ័ត្នចំពោះការសន្យា ការអះអាងថាអាចព្យាបាលបាន ឬដំបូន្មានដែលប្រាប់ឲ្យគ្រប់គ្នាធ្វើតាមការព្យាបាល ឬផែនការដូចគ្នា។

## តើអាចរកការគាំទ្រពីកន្លែងណា?

សួរអ្នកដែលមានជំងឺថាតើប្រភេទការគាំទ្រណាដែលអាចជួយខ្លួន និងចង់ឲ្យនរណាចូលរួម។ មិត្តភក្តិ ឬសមាជិកគ្រួសារដែលខ្លួនទុកចិត្តអាចជាការសន្ទនាដំបូងដ៏ល្អ។ អ្នកខ្លះចូលចិត្តក្រុមគាំទ្រ អ្នកប្រឹក្សា អង្គការជំងឺភ្លេចភ្លាំងក្នុងតំបន់ ឬការសន្ទនាជាមួយក្រុមថែទាំរបស់ខ្លួន។ ជម្រើសត្រឹមត្រូវអាស្រ័យលើអ្នកម្នាក់ៗ និងអ្វីដែលមាននៅក្បែរនោះ។

ការគាំទ្រមានសម្រាប់អ្នកដែលត្រូវបានធ្វើរោគវិនិច្ឆ័យ ហើយក៏សម្រាប់គ្រួសារ និងមិត្តភក្តិដែរ។ អ្នកថែទាំអាចត្រូវការពេលសម្រាក ឬជំនួយក្នុងកិច្ចការប្រចាំថ្ងៃ ខណៈអ្នកដែលមានជំងឺអាចចង់បានព័ត៌មាន ឬឱកាសជួបអ្នកដទៃដែលរស់នៅជាមួយជំងឺភ្លេចភ្លាំង។ តម្រូវការទាំងនេះអាចខុសគ្នា។ សូមសួរមុនពេលរៀបចំក្រុម ណាត់ជួប ឬចែករំលែកព័ត៌មានផ្ទាល់ខ្លួន។

[មគ្គុទ្ទេសក៍ថែទាំខ្លួនឯងសម្រាប់អ្នកថែទាំ](/caregiving/caregiver-self-care/) របស់យើងមានគំនិតសម្រាប់ការសុំឲ្យអ្នកដទៃជួយចែករំលែកការគាំទ្រ។ ប្រសិនបើអ្នកដែលមានជំងឺចង់បានជំនួយបន្ថែមក្នុងពេលថ្ងៃ មគ្គុទ្ទេសក៍[កម្មវិធីថែទាំប្រចាំថ្ងៃសម្រាប់មនុស្សពេញវ័យ](/caregiving/choosing-adult-day-program-dementia/) និង[ការថែទាំនៅផ្ទះ](/caregiving/choosing-in-home-dementia-care/) ពន្យល់ពីសំណួរដែលគួរសួរអំពីសេវាក្នុងតំបន់។

## តើអ្នកដែលមានជំងឺអាចនៅតែចូលរួមក្នុងការសម្រេចចិត្តដោយរបៀបណា?

ការធ្វើរោគវិនិច្ឆ័យតែមួយមុខមិនបានប្រាប់ថាអ្នកម្នាក់អាចសម្រេចអ្វីបាន ឬខ្លួនចង់ចូលរួមយ៉ាងណាឡើយ។ សូមសួរអ្នកនោះដោយផ្ទាល់។ ផ្តល់ពេលឲ្យពួកគេ ពន្យល់ជម្រើសដោយភាសាសាមញ្ញ ហើយសួរថាការគាំទ្រណាអាចធ្វើឲ្យការសន្ទនាងាយស្រួលជាងមុន។ ជម្រើសមួយចំនួនអាចពិភាក្សាបានងាយស្រួលជាងនៅកន្លែងស្ងាត់ ឬក្នុងការសន្ទនាជាច្រើនដង។

ប្រសិនបើអ្នកដែលមានជំងឺចង់រៀបចំសម្រាប់ការថែទាំសុខភាព ហិរញ្ញវត្ថុ ឬកន្លែងរស់នៅនាពេលអនាគត សូមចាប់ផ្តើមពីអ្វីដែលសំខាន់ចំពោះពួកគេ និងអ្នកដែលពួកគេទុកចិត្ត។ [មគ្គុទ្ទេសក៍ការរៀបចំរបស់ Dementia Australia](https://www.dementia.org.au/living-dementia/what-next-after-your-diagnosis/planning-ahead) ផ្តល់យោបល់ឲ្យពិភាក្សាបំណងប្រាថ្នារបស់ខ្លួនជាមួយមនុស្សដែលខ្លួនទុកចិត្ត និងអ្នកជំនាញពាក់ព័ន្ធ។ ច្បាប់ឯកសារ និងច្បាប់ស្តីពីការសម្រេចចិត្តខុសគ្នាតាមតំបន់ ដូច្នេះសូមសុំយោបល់ពីអ្នកជំនាញក្នុងតំបន់ដែលមានសមត្ថភាព មុនពេលពឹងផ្អែកលើមគ្គុទ្ទេសក៍ទូទៅ។ សម្រាប់ការជួយបង់វិក្កយបត្រប្រចាំថ្ងៃ និងបញ្ហាការក្លែងបន្លំ សូមមើល[មគ្គុទ្ទេសក៍គ្រប់គ្រងលុយជាមួយជំងឺភ្លេចភ្លាំង](/caregiving/managing-money-with-dementia/) របស់យើង។

### តើយើងអាចសួរអ្វីខ្លះក្នុងការសន្ទនាដំបូងអំពីការថែទាំនាពេលអនាគត?

សួរថាអ្នកដែលមានជំងឺចង់និយាយឥឡូវនេះឬទេ ហើយលើកយកប្រធានបទម្តងមួយ។ [សមាគម Alzheimer's Association ណែនាំការសន្ទនាតូចៗជាច្រើនដង ភាសាសាមញ្ញ និងការស្តាប់បំណងប្រាថ្នា ជំនឿ និងតម្លៃរបស់អ្នកដែលមានជំងឺ](https://www.alz.org/professionals/health-systems-medical-professionals/management/advanced-care-planning)។ [Alzheimer's Society បញ្ជាក់ថាមនុស្សម្នាក់ៗមានភាពត្រៀមខ្លួនខុសគ្នាក្នុងការគិតទុកជាមុន](https://www.alzheimers.org.uk/get-support/living-with-dementia/planning-ahead)។ វាល្អណាស់ក្នុងការឈប់ និងត្រឡប់ទៅនិយាយម្តងទៀតនៅថ្ងៃផ្សេង។

ប្រសិនបើអ្នកដែលមានជំងឺចង់បានសំណួរនាំផ្លូវ សូមសាកល្បងមួយក្នុងចំណោមនេះ៖

- “តើអ្នកចង់រក្សាផ្នែកណានៃថ្ងៃរបស់អ្នកបំផុត?”
- “តើអ្នកចង់ឲ្យនរណាចូលរួម ពេលយើងនិយាយអំពីការថែទាំនាពេលអនាគត?”
- “តើអ្វីជួយឲ្យអ្នកមានអារម្មណ៍ស្រួល និងត្រូវបានគោរព ពេលអ្នកត្រូវការជំនួយ?”
- “តើមានអ្វីដែលអ្នកចង់ឲ្យក្រុមថែទាំដឹងអំពីអ្វីដែលសំខាន់ចំពោះអ្នកទេ?”
- “តើអ្នកចង់ឲ្យខ្ញុំកត់ត្រាអ្វីដែលអ្នកបាននិយាយ ហើយនរណាគួរឃើញវាទេ?”

ស្តាប់ដោយមិនទាន់ស្នើឲ្យសម្រេចចិត្តភ្លាមៗ។ [វិទ្យាស្ថានជាតិស្តីពីភាពចាស់ (National Institute on Aging) ពិពណ៌នាថាការសន្ទនាដែលមានន័យអំពីតម្លៃ និងចំណូលចិត្ត ជាចំណុចសំខាន់នៃការរៀបចំការថែទាំមុនពេលចាំបាច់](https://order.nia.nih.gov/sites/default/files/2024-03/caring-for-a-Person-with-alzheimers-disease.pdf)។ ប្រសិនបើអ្នកដែលមានជំងឺយល់ព្រម សូមកត់ពាក្យរបស់ពួកគេ កាលបរិច្ឆេទ និងអ្នកដែលនៅទីនោះ។ សូមសុំការអនុញ្ញាតមុនពេលចែករំលែកកំណត់ចំណាំនេះ។ កំណត់ចំណាំទាំងនេះអាចជួយការសន្ទនាពេលក្រោយ ប៉ុន្តែមិនមែនជាឯកសារច្បាប់ ឬបញ្ជាវេជ្ជសាស្ត្រទេ។

កុំធ្វើការសន្មត់ដោយផ្អែកតែលើការធ្វើរោគវិនិច្ឆ័យ ឬការសន្ទនាតែមួយដង។ បំណងប្រាថ្នារបស់អ្នកដែលមានជំងឺអាចផ្លាស់ប្តូរ ហើយពួកគេអាចស្នើឲ្យផ្អាក ឬត្រឡប់ទៅប្រធានបទនេះនៅពេលក្រោយ។ សូមឲ្យពួកគេចូលរួមក្នុងការសម្រេចចិត្តដែលប៉ះពាល់ដល់ពួកគេ តាមកម្រិតដែលពួកគេចង់។

## តើនរណាគួរដឹងអំពីការធ្វើរោគវិនិច្ឆ័យ?

អ្នកដែលមានជំងឺជាអ្នកសម្រេចថាតើនឹងប្រាប់អ្នកដទៃឬទេ នៅពេលណា និងយ៉ាងណា លុះត្រាតែមានស្ថានភាពសុវត្ថិភាព ឬច្បាប់ជាក់លាក់ដែលត្រូវការការណែនាំពីអ្នកជំនាញ។ សួរថាពួកគេចង់ប្រាប់នរណា និងចង់ឲ្យអ្នកទាំងនោះដឹងអ្វី។ ពួកគេអាចជ្រើសរើសមនុស្សដែលទុកចិត្តម្នាក់ជាមុនសិន ឬមិនចង់ចែករំលែកឲ្យទូលំទូលាយ។

ប្រសិនបើអ្នកដែលមានជំងឺចង់បានជំនួយក្នុងការនិយាយ សូមសួរថាពួកគេចង់ឲ្យអ្នកនិយាយអ្វី និងថាតើពួកគេចង់នៅទីនោះឬទេ។ ចែករំលែកតែព័ត៌មានដែលពួកគេបានយល់ព្រមចែករំលែកប៉ុណ្ណោះ។ [មគ្គុទ្ទេសក៍ការទំនាក់ទំនង](/caregiving/communication-tips/) របស់យើងគ្របដណ្តប់លើការសន្ទនាដោយគោរពជាមួយគ្រួសារ និងអ្នកផ្សេងទៀត។

## តើជីវិតប្រចាំថ្ងៃអាចបន្តបានដោយរបៀបណាខណៈយើងកំពុងសម្របខ្លួន?

រក្សាទម្លាប់ និងសកម្មភាពដែលសំខាន់ចំពោះអ្នកដែលមានជំងឺ ក្នុងកម្រិតដែលពួកគេនៅតែមានអារម្មណ៍ស្រួល។ ការធ្វើរោគវិនិច្ឆ័យមិនមានន័យថាចំណូលចិត្តគ្រប់យ៉ាង ការដើរលេង ឬតួនាទីក្នុងគ្រួសារត្រូវឈប់ទាំងអស់ទេ។ ប្រសិនបើអ្វីមួយកាន់តែពិបាក សូមសួរថាតើពួកគេចង់ផ្លាស់ប្តូរអ្វី។ ការមកលេងខ្លីៗ ភ្លេងដែលពួកគេស្គាល់ ជំនួយក្នុងជំហានមួយ ឬពេលថ្ងៃផ្សេងអាចសមនឹងពួកគេជាង។

អ្នកដែលមានជំងឺក៏អាចចង់មានពេលសម្រាប់ទទួលយកព័ត៌មាននេះដោយមិនពិភាក្សាផែនការដែរ។ សូមបន្តសួរសុខទុក្ខ ដោយមិនធ្វើឲ្យគ្រប់ការសន្ទនាក្លាយជាកិច្ចប្រជុំថែទាំ។ ការសម្រាកពីការរៀបចំ ហើយចំណាយពេលរួមគ្នាតាមរបៀបធម្មតា ក៏ជាការល្អដែរ។

## តើពេលណាគួរសុំការណែនាំពីអ្នកជំនាញ?

សួរគ្រូពេទ្យរបស់អ្នកដែលមានជំងឺអំពីសំណួរជាក់លាក់ទាក់ទងនឹងការធ្វើរោគវិនិច្ឆ័យ សុខភាព ឱសថ ឬរោគសញ្ញារបស់ពួកគេ។ សម្រាប់ការសម្រេចចិត្តផ្នែកច្បាប់ និងហិរញ្ញវត្ថុ សូមនិយាយជាមួយអ្នកជំនាញដែលមានសមត្ថភាព ហើយយល់ច្បាប់នៅកន្លែងដែលអ្នកដែលមានជំងឺរស់នៅ។ គេហទំព័រទូទៅមិនអាចប្រាប់អ្នកបានថាការព្យាបាល ឯកសារ ឬសេវាណាត្រូវតាមអ្នកម្នាក់ៗឡើយ។

ការភ្ងេរច្រឡំភ្លាមៗ ឬការផ្លាស់ប្តូរសុខភាពភ្លាមៗផ្សេងទៀត ត្រូវការការថែទាំវេជ្ជសាស្ត្របន្ទាន់។ [ការណែនាំរបស់ NHS អំពីការភ្ងេរច្រឡំភ្លាមៗ](https://www.nhs.uk/symptoms/confusion/) បញ្ជាក់ថាត្រូវទទួលការជួយពីគ្រូពេទ្យភ្លាមៗ។ សូមប្រើប្រព័ន្ធសង្គ្រោះបន្ទាន់ ឬប្រព័ន្ធថែទាំបន្ទាន់នៅកន្លែងដែលអ្នកដែលមានជំងឺរស់នៅ ហើយកុំសន្មតថាការផ្លាស់ប្តូរភ្លាមៗជាផ្នែកមួយនៃស្ថានភាពធម្មតារបស់ពួកគេ។

ប្រសិនបើការរៀបចំមានអារម្មណ៍ធ្ងន់ធ្ងរ សូមជ្រើសរើសមនុស្សដែលទុកចិត្តម្នាក់ ឬសេវាមួយដើម្បីជួយរៀបចំសំណួរជាប្រភេទភ្លាមៗ និងប្រភេទនៅពេលក្រោយ។ អ្នកដែលមានជំងឺអាចកំណត់ល្បឿន និងសម្រេចថាចង់ឲ្យជួយអ្វី។ ការធ្វើរោគវិនិច្ឆ័យជាព័ត៌មានសំខាន់ ប៉ុន្តែវាមិនកំណត់ចំណាប់អារម្មណ៍ ទំនាក់ទំនង ឬជីវិតប្រចាំថ្ងៃរបស់អ្នកដែលមានជំងឺឡើយ។
